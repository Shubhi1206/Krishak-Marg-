// Language selection component
function LanguageButton({ currentLang, onSelect }) {
  const [visible, setVisible] = useState(false);
  const languages = [
    { code: 'en', label: 'English' },
    { code: 'hi', label: 'हिन्दी' },
    { code: 'pa', label: 'ਪੰਜਾਬੀ' },
  ];

  return (
    <View>
      <TouchableOpacity style={styles.languageButton} onPress={() => setVisible(true)}>
        <Icon name="language" size={22} color="#0e8543" />
        <Text style={{marginLeft: 6}}>{languages.find(l => l.code === currentLang)?.label}</Text>
      </TouchableOpacity>
      <Modal visible={visible} transparent animationType="fade">
        <TouchableOpacity style={styles.languageModalBG} onPress={() => setVisible(false)}>
          <View style={styles.languageModal}>
            {languages.map(l => (
              <TouchableOpacity
                key={l.code}
                style={styles.languageOption}
                onPress={() => {
                  onSelect(l.code);
                  setVisible(false);
                }}>
                <Text style={{fontSize: 17}}>{l.label}</Text>
              </TouchableOpacity>
            ))}
          </View>
        </TouchableOpacity>
      </Modal>
    </View>
  );
}

const Tab = createMaterialBottomTabNavigator();

export default function App() {
  const { i18n } = useTranslation();
  const [lang, setLang] = useState(i18n.language);

  function handleLangChange(langCode) {
    setLang(langCode);
    i18n.changeLanguage(langCode);
  }

  return (
    <NavigationContainer>
      <Tab.Navigator
        initialRouteName="Home"
        activeColor="#0e8543"
        barStyle={{ backgroundColor: 'white', borderTopColor: '#eee', borderTopWidth: 1 }}>
        <Tab.Screen
          name="Home"
          options={{
            tabBarLabel: 'Home',
            tabBarIcon: ({ color }) => (
              <Icon name="home" color={color} size={24} />
            ),
          }}
        >
          {() => <HomeScreen onLangChange={handleLangChange} lang={lang} />}
        </Tab.Screen>
        {/* Add other tabs similarly if desired */}
      </Tab.Navigator>
    </NavigationContainer>
  );
}
